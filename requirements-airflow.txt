# Extra packages baked into the Airflow image on top of apache/airflow:2.8.4.
#
# The DAG imports the `pipeline` package, which needs scikit-learn and mlflow.
# Mounting the DAG into the stock Airflow image without them is the classic
# Lab 2 failure: the scheduler parses the DAG, the import raises, and the UI
# shows a broken DAG whose traceback says nothing about the real cause.
#
# Two different rules apply to the lines below, and it matters which is which.
#
# PINNED BY AIRFLOW. numpy, pandas and pyarrow appear in Airflow 2.8.4's own
# constraints file, so --constraint decides them and you cannot match the
# versions in requirements.txt. This is the price of sharing an interpreter
# with Airflow, and the reason the local environment and the Airflow image are
# two separate dependency sets.
numpy==1.24.4
pandas==2.1.4
pyarrow==14.0.2

# CHOSEN BY US. scikit-learn, joblib and mlflow are NOT in Airflow's
# constraints file, so we pick them — and they must match requirements.txt
# exactly. A model pickled by scikit-learn 1.3.2 inside this image does not
# load in an environment running 1.6.0; it fails with
#   AttributeError: 'DecisionTreeClassifier' object has no attribute 'monotonic_cst'
# That is a real failure, not a warning, and it is silent until something
# downstream tries to load the model this pipeline registered in MLflow.
scikit-learn==1.6.0
joblib==1.4.2
mlflow==2.19.0
